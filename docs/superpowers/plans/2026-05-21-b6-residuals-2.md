@router.post("/{workspace_id}/turns/{turn_job_id}/cancel")
def cancel_turn(
    workspace_id: uuid.UUID,
    turn_job_id: uuid.UUID,
    request: Request,
    user=Depends(get_current_user),  # noqa: B008
    session: Session = Depends(_session_dep),  # noqa: B008
) -> HTMLResponse:
    """Mark queued turn_job as 'cancelled'. No-op if status != 'queued'.

    Race-safe: the WHERE clause includes status='queued', so a concurrent
    processor that has already moved it to 'running' wins; this handler
    returns the noop message.

    Audit: on successful cancel, writes a `cloud_turn_cancelled`
    WorkspaceEvent in the same transaction. Without this, the
    submit-time `cloud_turn_sent` row has no matching outcome event
    and the compliance chain breaks (see B6-residuals review).
    """
    from datetime import UTC, datetime

    ws = _require_owner(
        workspace_id=workspace_id,
        user=user,
        session=session,
    )
    result = session.execute(
        update(TurnJob)
        .where(
            TurnJob.id == turn_job_id,
            TurnJob.workspace_id == ws.id,
            TurnJob.status == "queued",
        )
        .values(status="cancelled", completed_at=datetime.now(UTC))
    )
    if result.rowcount:
        session.add(
            WorkspaceEvent(
                workspace_id=ws.id,
                kind="cloud_turn_cancelled",
                actor_id=user.id,
                payload={
                    "turn_job_id": str(turn_job_id),
                    "tokens_in": 0,
                    "tokens_out": 0,
                },
            )
        )
    session.commit()
    if result.rowcount:
        return HTMLResponse("<p>cancelled</p>")
    return HTMLResponse("<p>too late — already started or finished</p>")
```

import asyncio

import pytest


@pytest.mark.asyncio
async def test_concurrent_submits_cannot_overshoot_cap(
    monkeypatch, pg_url, pg_session_factory
):
    """Two concurrent submits, each individually fine, must not both
    succeed if together they'd exceed the daily cap. The cap check +
    insert must serialize via row-level lock on workspaces.

    Without the SELECT ... FOR UPDATE on the workspaces row, both
    submits would read used_in=60 and both pass the gate of 100."""
    from concurrent.futures import ThreadPoolExecutor
    from datetime import UTC, datetime, timedelta

    from call_manager.config import Settings
    from call_manager.db.models import ChatTurn
    from call_manager.proposal_assist.turns.caps import TokenCapExceeded
    from call_manager.proposal_assist.turns.submit import submit_user_turn
    from tests.proposal_assist.conftest import make_user, make_workspace

    monkeypatch.setenv("DATABASE_URL", pg_url)
    monkeypatch.setenv("SESSION_SECRET", "test-secret-at-least-16-chars")
    monkeypatch.setenv("ALLOWED_EMAIL_DOMAINS", "ki.se")
    monkeypatch.setenv("APP_BASE_URL", "http://testserver")
    monkeypatch.setenv("DAILY_TOKEN_CAP_INPUT", "100")
    monkeypatch.setenv("DAILY_TOKEN_CAP_OUTPUT", "1000000")
    settings = Settings()

    with pg_session_factory() as s:
        owner = make_user(s)
        s.commit()
        ws = make_workspace(s, owner=owner)
        s.commit()
        # Seed: 60 input tokens used today. Cap is 100. A single turn
        # would pass; two concurrent turns would both read 60 and both
        # pass the gate without the row lock.
        s.add(
            ChatTurn(
                workspace_id=ws.id,
                role="assistant",
                content={"type": "text", "text": "x"},
                tokens_in=60,
                created_at=datetime.now(UTC) - timedelta(minutes=5),
            )
        )
        s.commit()
        wsid = ws.id
        owner_id = owner.id

    def attempt():
        with pg_session_factory() as session:
            try:
                submit_user_turn(
                    session=session,
                    workspace_id=wsid,
                    content={"type": "text", "text": "hi"},
                    submitted_by_user_id=owner_id,
                    settings=settings,
                )
                session.commit()
                return "ok"
            except TokenCapExceeded:
                session.rollback()
                return "capped"

    loop = asyncio.get_running_loop()
    with ThreadPoolExecutor(max_workers=2) as ex:
        results = await asyncio.gather(
            loop.run_in_executor(ex, attempt),
            loop.run_in_executor(ex, attempt),
        )

    assert sorted(results) == ["capped", "ok"], (
        f"expected one ok and one capped, got {results}"
    )
```

        .with_for_update()

    cap = check_daily_token_cap(session, workspace_id, settings=settings)
