# Ranking v2.1 Improvements Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the ranking score continuous (logprob expected-value), order ties by cosine, replace the cosine-based Sonnet router with outcome-based escalation backed by an append-only attempt ledger, stand up a DB-backed golden set with admin curation, and feed org-profile structured fields into embeddings.

**Architecture:** Five independently shippable phases over the existing `ranking/` + `embeddings/` + `digest/` + `web/` modules. No new ranking scheme; same pipeline shape (gates → local LLM → optional escalation → aggregate). Each phase lands with green tests and depends on no later phase. Migrations run 0042→0045.

**Tech Stack:** Python 3.12, FastAPI + Jinja2, SQLAlchemy 2.x + Alembic, PostgreSQL 16 + pgvector, Ollama (Gemma 4 E4B local) + Anthropic (Sonnet remote), pytest + testcontainers.

**Spec:** [docs/superpowers/specs/2026-06-11-ranking-v2.1-improvements-design.md](../specs/2026-06-11-ranking-v2.1-improvements-design.md)

# src/call_manager/embeddings/calibration.py
"""Embedding-model → threshold calibration registry + startup guard.

The cosine pre-gate thresholds (EMBEDDING_PREGATE_THRESHOLD /
EMBEDDING_HARD_FAIL_THRESHOLD) were calibrated against BGE-M3's similarity
distribution. A different embedding model has a different distribution, so
the same numeric cutoffs mean something different. This module warns when a
threshold still rests on the bge-m3 default while an unrecognized model is
configured. It never refuses to start — see spec §3.6.
"""

from __future__ import annotations

# The bge-m3-calibrated defaults, mirrored from config.py's Field defaults.
# Keep in lockstep with Settings.embedding_pregate_threshold /
# embedding_hard_fail_threshold.
BGE_M3_DEFAULTS = {"pregate": 0.30, "hard_fail": 0.20}

# Models whose calibration we have validated. Add an entry (with its own
# default pair) when a new model's distribution is characterized.
THRESHOLD_CALIBRATIONS: dict[str, dict[str, float]] = {
    "bge-m3": BGE_M3_DEFAULTS,
}


def check_threshold_calibration(
    *, model: str, pregate: float, hard_fail: float
) -> str | None:
    """Return a warning string when `model` is unrecognized AND at least one
    threshold still equals its bge-m3 default; otherwise None.

    Per-threshold (R8): an operator who overrides only `pregate` still gets
    warned that `hard_fail` is uncalibrated, and vice versa.
    """
    if model in THRESHOLD_CALIBRATIONS:
        return None
    stale: list[str] = []
    if pregate == BGE_M3_DEFAULTS["pregate"]:
        stale.append("EMBEDDING_PREGATE_THRESHOLD")
    if hard_fail == BGE_M3_DEFAULTS["hard_fail"]:
        stale.append("EMBEDDING_HARD_FAIL_THRESHOLD")
    if not stale:
        return None
    return (
        f"Embedding model {model!r} has no validated threshold calibration; "
        f"these settings still rest on bge-m3 defaults and likely mis-gate: "
        f"{', '.join(stale)}. Set them explicitly for {model!r}."
    )
```

def _append_history(report: GoldenSetReport, version_counts: dict[str, int]) -> None:
    """Append one JSONL line per run so P3/P4 before/after tau stays readable.
    Best-effort: a write failure must never fail the eval tick."""
    from datetime import UTC, datetime

    line = json.dumps({
        "timestamp": datetime.now(UTC).isoformat(),
        "n_fixture": report.n_fixture,
        "n_compared": report.n_compared,
        "tau": report.tau,
        "prompt_version_counts": version_counts,
    })
    try:
        _HISTORY_PATH.parent.mkdir(parents=True, exist_ok=True)
        with _HISTORY_PATH.open("a", encoding="utf-8") as fh:
            fh.write(line + "\n")
    except OSError as e:
        _logger.warning(
            "ranking.golden_eval.history_write_failed",
            extra={"event": "golden_history_write_failed", "error": str(e)},
        )
```

Session; the web route owns the transaction (commit/rollback)."""

from __future__ import annotations

from dataclasses import dataclass

from sqlalchemy import and_, exists, func, or_, select

from call_manager.db.models import (
    Call, CallRanking, GoldenSetTriple, Profile, Subscription, UserCallDismissal,
)

_CANDIDATE_LIMIT = 20


@dataclass(frozen=True)
class TripleRow:
    id: int
    call_id: int
    profile_id: int
    expected_score: int
    note: str | None
    call_title: str
    profile_name: str


@dataclass(frozen=True)
class CandidateRow:
    call_id: int
    profile_id: int
    call_title: str
    profile_name: str
    aggregate_score: int
    embed_similarity: float | None
    reason: str  # which mining signal surfaced it


def add_triple(session, *, call_id, profile_id, expected_score, note=None, created_by=None) -> int:
    row = GoldenSetTriple(
        call_id=call_id, profile_id=profile_id, expected_score=expected_score,
        note=note, created_by=created_by,
    )
    session.add(row)
    session.flush()
    return row.id


def delete_triple(session, *, triple_id: int) -> None:
    row = session.get(GoldenSetTriple, triple_id)
    if row is not None:
        session.delete(row)


def list_triples(session) -> list[TripleRow]:
    rows = session.execute(
        select(GoldenSetTriple, Call.title, Profile.name)
        .join(Call, Call.id == GoldenSetTriple.call_id)
        .join(Profile, Profile.id == GoldenSetTriple.profile_id)
        .order_by(GoldenSetTriple.created_at.desc())
    ).all()
    return [
        TripleRow(id=t.id, call_id=t.call_id, profile_id=t.profile_id,
                  expected_score=t.expected_score, note=t.note,
                  call_title=title, profile_name=name)
        for t, title, name in rows
    ]


def _not_already_a_triple():
    """Correlated NOT EXISTS against golden_set_triples for a CallRanking row."""
    return ~exists().where(
        and_(
            GoldenSetTriple.call_id == CallRanking.call_id,
            GoldenSetTriple.profile_id == CallRanking.profile_id,
        )
    )


def candidates_dismissed_high_score(session) -> list[CandidateRow]:
    """Ranked >= 60 for a profile, dismissed by a user subscribed to that
    profile. UserCallDismissal is per (user, call); we require the dismissing
    user to subscribe to the ranking's profile."""
    q = (
        select(CallRanking, Call.title, Profile.name)
        .join(Call, Call.id == CallRanking.call_id)
        .join(Profile, Profile.id == CallRanking.profile_id)
        .where(
            CallRanking.aggregate_score >= 60,
            _not_already_a_triple(),
            exists().where(
                and_(
                    UserCallDismissal.call_id == CallRanking.call_id,
                    Subscription.user_id == UserCallDismissal.user_id,
                    Subscription.profile_id == CallRanking.profile_id,
                )
            ),
        )
        .order_by(CallRanking.aggregate_score.desc())
        .limit(_CANDIDATE_LIMIT)
    )
    return [
        CandidateRow(call_id=r.call_id, profile_id=r.profile_id, call_title=title,
                     profile_name=name, aggregate_score=r.aggregate_score,
                     embed_similarity=r.embed_similarity, reason="dismissed_high_score")
        for r, title, name in session.execute(q).all()
    ]


def candidates_cosine_disagreement(session) -> list[CandidateRow]:
    """High cosine + low score, or low cosine + high score."""
    q = (
        select(CallRanking, Call.title, Profile.name)
        .join(Call, Call.id == CallRanking.call_id)
        .join(Profile, Profile.id == CallRanking.profile_id)
        .where(
            CallRanking.embed_similarity.is_not(None),
            or_(
                and_(CallRanking.embed_similarity >= 0.5, CallRanking.aggregate_score <= 20),
                and_(CallRanking.embed_similarity < 0.3, CallRanking.aggregate_score >= 60),
            ),
            _not_already_a_triple(),
        )
        .order_by(CallRanking.embed_similarity.desc())
        .limit(_CANDIDATE_LIMIT)
    )
    return [
        CandidateRow(call_id=r.call_id, profile_id=r.profile_id, call_title=title,
                     profile_name=name, aggregate_score=r.aggregate_score,
                     embed_similarity=r.embed_similarity, reason="cosine_disagreement")
        for r, title, name in session.execute(q).all()
    ]

    """
    return []

import math

from call_manager.ranking.ev import extract_topic_expected


def _char_tokens(content: str, *, rating_top_logprobs: list[dict], rating_char: str):
    """Build a realistic Ollama logprobs envelope by char-tokenizing `content`
    (one token per character, each with its UTF-8 bytes). At the FIRST
    occurrence of `rating_char` that is preceded by `"rating":` we attach
    `rating_top_logprobs`; every other token gets a trivial single-entry
    top_logprobs. This mirrors the real envelope's shape closely enough to
    exercise the offset binding."""
    content_tokens = []
    marker = '"rating":'
    rating_pos = content.index(marker) + len(marker)
    # advance past spaces to the digit
    while content[rating_pos] == " ":
        rating_pos += 1
    for i, ch in enumerate(content):
        tok = {"token": ch, "bytes": list(ch.encode("utf-8")), "logprob": -0.01}
        if i == rating_pos and ch == rating_char:
            tok["top_logprobs"] = rating_top_logprobs
        else:
            tok["top_logprobs"] = [{"token": ch, "logprob": -0.01,
                                    "bytes": list(ch.encode("utf-8"))}]
        content_tokens.append(tok)
    return {
        "choices": [{
            "index": 0,
            "message": {"role": "assistant", "content": content},
            "logprobs": {"content": content_tokens},
        }]
    }


def test_expected_value_weights_digits():
    content = '{"topic_fit": {"rating": 4, "rationale": "good"}}'
    top = [
        {"token": "4", "logprob": math.log(0.7), "bytes": [52]},
        {"token": "3", "logprob": math.log(0.3), "bytes": [51]},
    ]
    env = _char_tokens(content, rating_top_logprobs=top, rating_char="4")
    ev = extract_topic_expected(env, 4)
    assert ev is not None
    assert abs(ev - (4 * 0.7 + 3 * 0.3)) < 1e-6  # 3.7

"""G-Eval expected-value extraction for the topic_fit rating.

Converts the LLM's single integer topic_fit rating into an expected value
over the rating token's top-k logprobs, recovering a continuous score from a
6-point Likert scale. Deterministic JSON-offset binding (Codex R7) locates
the structural `topic_fit.rating` literal so we never weight a digit that
happens to appear in rationale prose. Any inconsistency returns None and the
caller falls back to argmax — this function never raises.
"""

from __future__ import annotations

import math

_DIGITS = {"0", "1", "2", "3", "4", "5"}


def extract_topic_expected(envelope: dict, parsed_rating: int) -> float | None:
    try:
        choice = envelope["choices"][0]
        content: str = choice["message"]["content"]
        tokens = choice["logprobs"]["content"]
    except (KeyError, IndexError, TypeError):
        return None
    if not isinstance(tokens, list) or not content:
        return None

    rating_offset = _locate_rating_literal_offset(content)
    if rating_offset is None:
        return None

    covering = _token_covering_offset(tokens, rating_offset)
    if covering is None:
        return None
    # Sanity: the covering token must be exactly the parsed rating digit.
    if covering.get("token") != str(parsed_rating):
        return None

    top = covering.get("top_logprobs")
    if not isinstance(top, list):
        return None
    weights: dict[int, float] = {}
    for entry in top:
        tok = entry.get("token") if isinstance(entry, dict) else None
        lp = entry.get("logprob") if isinstance(entry, dict) else None
        if tok in _DIGITS and isinstance(lp, (int, float)):
            weights[int(tok)] = weights.get(int(tok), 0.0) + math.exp(lp)
    if len(weights) < 2:
        return None  # degenerate — EV == argmax, not worth recording
    total = sum(weights.values())
    if total <= 0:
        return None
    ev = sum(d * w for d, w in weights.items()) / total
    if not math.isfinite(ev) or not 0.0 <= ev <= 5.0:
        return None
    return ev


def _token_covering_offset(tokens: list, target: int) -> dict | None:
    """Walk tokens accumulating character offsets (from each token's UTF-8
    bytes, falling back to its text) and return the token whose span covers
    `target`."""
    pos = 0
    for tok in tokens:
        if not isinstance(tok, dict):
            return None
        raw = tok.get("bytes")
        if isinstance(raw, list):
            try:
                text = bytes(raw).decode("utf-8")
            except (ValueError, TypeError):
                text = tok.get("token", "")
        else:
            text = tok.get("token", "")
        end = pos + len(text)
        if pos <= target < end:
            return tok
        pos = end
    return None


def _locate_rating_literal_offset(content: str) -> int | None:
    """Return the character offset of the integer literal that is the VALUE of
    the structural `topic_fit.rating` key. String-literal-aware: ignores any
    `"rating"` text that sits inside a JSON string value.

    Minimal scanner: walk the chars tracking whether we're inside a string
    (respecting backslash escapes). Outside strings, detect the key sequence
    `"topic_fit"` then, after it, the key `"rating"` followed by `:` and an
    integer literal; return the digit's offset.
    """
    n = len(content)
    i = 0
    in_string = False
    escape = False
    saw_topic_fit = False
    # Track the most recent completed string-literal (key candidate).
    last_key: str | None = None
    key_start = -1
    while i < n:
        ch = content[i]
        if in_string:
            if escape:
                escape = False
            elif ch == "\\":
                escape = True
            elif ch == '"':
                in_string = False
                last_key = content[key_start + 1 : i]
            i += 1
            continue
        if ch == '"':
            in_string = True
            key_start = i
            i += 1
            continue
        if ch == ":" and last_key is not None:
            key = last_key
            last_key = None
            if key == "topic_fit":
                saw_topic_fit = True
            elif key == "rating" and saw_topic_fit:
                # Skip whitespace to the integer literal.
                j = i + 1
                while j < n and content[j] in " \t\n\r":
                    j += 1
                if j < n and content[j].isdigit():
                    return j
                return None
        i += 1
    return None
```

# src/call_manager/ranking/validate_ev.py
"""Non-persisting EV validation. Re-runs the local rubric on a sample of
already-ranked pairs, computes the three P3 acceptance metrics, writes a
timestamped JSON report under CALL_MANAGER_DATA_DIR. Zero DB writes — this
gates the v5.0 version flip without touching production scores (spec §3.1).

    uv run python -m call_manager.ranking.validate_ev --sample 50
"""

from __future__ import annotations

import argparse
import json
import logging
import os
import sys
from datetime import UTC, datetime
from pathlib import Path

from sqlalchemy import func, select

from call_manager.db.models import Call, CallRanking, Profile
from call_manager.ranking.aggregate import LLMResult, aggregate
from call_manager.ranking.criteria import evaluate_gates
from call_manager.ranking.ev import extract_topic_expected
from call_manager.ranking.prompt import build_messages, parse_response
from call_manager.ranking.rank_pair import _build_verification_text, _safe_cosine

_logger = logging.getLogger(__name__)


def run_validation(*, session_factory, client, sample: int, data_dir: Path) -> Path:
    """Sample up to `sample` LLM-scored pairs (stratified by score), re-run the
    local rubric with logprobs in memory, compare EV-score vs argmax-score.
    Returns the path of the written report. No DB writes."""
    rows: list[tuple[int, int]] = []
    with session_factory() as session:
        # Stratify: order by score so the sample spans the range, then take a
        # bounded slice. Only pairs that actually ran the LLM (have a non-null
        # model and weren't gate-zeroed) are meaningful.
        candidates = session.execute(
            select(CallRanking.call_id, CallRanking.profile_id)
            .where(CallRanking.model != "", CallRanking.aggregate_reason.is_(None))
            .order_by(CallRanking.aggregate_score)
            .limit(sample)
        ).all()
        rows = [(r.call_id, r.profile_id) for r in candidates]

    results = []
    now = datetime.now(UTC)
    for call_id, profile_id in rows:
        with session_factory() as session:
            call = session.get(Call, call_id)
            profile = session.get(Profile, profile_id)
            if call is None or profile is None:
                continue
            messages = build_messages(profile=profile, call=call, enable_analysis=False)
            embed_sim = _safe_cosine(profile, call)
            gates = evaluate_gates(profile, call, now=now, embed_similarity=embed_sim,
                                   embedding_threshold=0.30)
            try:
                envelope = client.chat(messages=messages,
                                       response_format={"type": "json_object"},
                                       logprobs=True, top_logprobs=10)
                content = envelope["choices"][0]["message"]["content"]
                parsed = parse_response(content,
                                        verification_text=_build_verification_text(call))
            except Exception as e:  # noqa: BLE001 — validation must not crash mid-run
                _logger.warning("validate_ev.pair_failed call=%s profile=%s err=%s",
                                call_id, profile_id, e)
                continue
            ev = extract_topic_expected(envelope, parsed.topic_fit_rating)
            argmax_score = aggregate(gates, LLMResult(parsed.topic_fit_rating,
                                                      parsed.eligibility_rating,
                                                      parsed.exclusions_rating)).score
            ev_score = aggregate(gates, LLMResult(parsed.topic_fit_rating,
                                                  parsed.eligibility_rating,
                                                  parsed.exclusions_rating,
                                                  topic_expected=ev)).score
            results.append({
                "call_id": call_id, "profile_id": profile_id,
                "argmax": parsed.topic_fit_rating, "ev": ev,
                "argmax_score": argmax_score, "ev_score": ev_score,
            })

    metrics = _metrics(results)
    data_dir.mkdir(parents=True, exist_ok=True)
    out = data_dir / f"ev_validation_{now.strftime('%Y%m%dT%H%M%S')}.json"
    out.write_text(json.dumps({"timestamp": now.isoformat(),
                               "n_sampled": len(results), **metrics,
                               "pairs": results}, indent=2))
    _logger.info("validate_ev.done n=%s metrics=%s", len(results), metrics)
    return out


def _metrics(results: list[dict]) -> dict:
    n = len(results)
    if n == 0:
        return {"populated_fraction": 0.0, "distinct_scores_40_100": 0,
                "within_one_point_fraction": 1.0}
    populated = sum(1 for r in results if r["ev"] is not None)
    distinct = len({r["ev_score"] for r in results if 40 <= r["ev_score"] <= 100})
    within = sum(1 for r in results
                 if r["ev"] is None or abs(r["ev"] - r["argmax"]) <= 1.0)
    return {
        "populated_fraction": populated / n,
        "distinct_scores_40_100": distinct,
        "within_one_point_fraction": within / n,
    }


def _build_client():
    from call_manager.config import Settings
    from call_manager.ranking.ollama_client import OllamaClient

    s = Settings()
    return OllamaClient(base_url=s.ollama_url, model=s.gemma_model)


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(prog="call_manager.ranking.validate_ev")
    parser.add_argument("--sample", type=int, default=50)
    args = parser.parse_args(argv)

    from call_manager.config import Settings
    from call_manager.db.session import make_engine, make_session_factory

    settings = Settings()
    engine = make_engine(settings.database_url)
    session_factory = make_session_factory(engine)
    data_dir = Path(os.environ.get("CALL_MANAGER_DATA_DIR", "data")).resolve()
    try:
        out = run_validation(session_factory=session_factory, client=_build_client(),
                             sample=args.sample, data_dir=data_dir)
    finally:
        engine.dispose()
    print(f"wrote {out}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

per attempt (any outcome) and counts today's rows to enforce the daily cap."""

from __future__ import annotations

from sqlalchemy import func, select
from sqlalchemy.orm import Session

from call_manager.db.models import RankingRemoteAttempt


def record_attempt(session: Session, *, call_id: int, profile_id: int, model: str,
                   outcome: str) -> None:
    session.add(RankingRemoteAttempt(
        call_id=call_id, profile_id=profile_id, model=model, outcome=outcome))
