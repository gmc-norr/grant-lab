# Ranking Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Score every (`call`, `profile`) pair via a locally-hosted Gemma 4 E4B (Ollama, OpenAI-compatible HTTP) into a `call_rankings` table; persist ranking jobs in `ranking_jobs` with retries and a dead-letter terminal state; auto-enqueue jobs when ingestion writes new calls; drain the queue from a background scheduler tick; surface scores on the dashboard via `/?profile=<id>`; expose queue depth and "re-rank everything" on `/admin`. After this plan ships, every ingested call is auto-ranked against every active profile and the dashboard becomes useful.

**Architecture:** New `ranking/` module exposes one job: `rank_pair(call, profile) → (score, rationale, raw_response)`. The job calls `OllamaClient` (a thin httpx wrapper around `/v1/chat/completions` with `response_format=json_object` and `temperature=0`). A new `Profile` model (full schema per spec, but no bootstrap UI yet — Plan 4) holds the ranking target. `ranking_jobs` is a Postgres-backed work queue keyed on `(call_id, profile_id)` with a partial unique index that prevents duplicate active jobs. The ingestion orchestrator gains one hook: after a successful run, enqueue jobs for the inserted calls × all active profiles. APScheduler runs a `drain_queue` job every 60 s that pops N pending jobs, runs them, marks them `done` or `failed`, dead-letters at 5 attempts.

**Tech Stack:** Python 3.12 · `httpx` (Ollama HTTP) · APScheduler (existing) · Postgres advisory queue via `SELECT … FOR UPDATE SKIP LOCKED` · `pytest` + `httpx.MockTransport` for client tests · existing foundation/ingestion modules

```
                             -- must_include[], hard_exclusions[]
  is_active bool default true,
  created_by FK users(id),
  created_at, updated_at

call_rankings
  call_id FK calls(id), profile_id FK profiles(id),
  score int 0-100, rationale text,
  model text, prompt_version text,
  raw_response jsonb,
  sample_count int default 1,
  ranked_at timestamptz
  PRIMARY KEY(call_id, profile_id)
  INDEX (profile_id, score DESC)        -- ranked feed sort

ranking_jobs
  id, call_id FK, profile_id FK,
  status enum('pending'|'running'|'failed'|'done'),
  attempts int default 0, last_error text,
  created_at, updated_at
  INDEX (status, updated_at)
  UNIQUE INDEX (call_id, profile_id) WHERE status IN ('pending', 'running')
```

The partial unique index is the contract that prevents stacking duplicate active jobs on retrigger / restart.

```python
from __future__ import annotations

import logging

import httpx

_logger = logging.getLogger(__name__)


class LLMError(Exception):
    """Wraps any failure from the LLM endpoint — network, HTTP, parse —
    so the ranking job processor can retry/backoff uniformly."""


class OllamaClient:
    """Thin HTTP client for Ollama's OpenAI-compatible /v1/chat/completions.

    Returns the raw OpenAI ChatCompletion dict; the caller is responsible
    for extracting and parsing the assistant message's JSON content. We
    keep the raw response so `call_rankings.raw_response` retains the full
    payload for debugging (per spec 'LLM reliability').

    `temperature=0` is hardcoded — the spec mandates it for run-to-run
    repeatability. `response_format={"type": "json_object"}` requests JSON
    mode; combined with Gemma 4's native function-calling/JSON support,
    this near-eliminates parse failures.

    Vendor swap: pointing this at OpenAI/Anthropic-compatible endpoints
    (Together, Groq, etc.) is a `base_url` + `model` config change.
    """

    def __init__(
        self,
        *,
        base_url: str,
        model: str,
        transport: httpx.BaseTransport | None = None,
        timeout: float = 60.0,
    ):
        self._base_url = base_url.rstrip("/")
        self.model = model  # public so rank_pair can fall back to it
        self._transport = transport
        self._timeout = timeout

    def chat(self, *, messages: list[dict]) -> dict:
        payload = {
            "model": self.model,
            "messages": messages,
            "temperature": 0,
            "response_format": {"type": "json_object"},
        }
        try:
            with httpx.Client(
                base_url=self._base_url,
                transport=self._transport,
                timeout=self._timeout,
            ) as client:
                resp = client.post("/v1/chat/completions", json=payload)
                resp.raise_for_status()
                return resp.json()
        except httpx.HTTPError as e:
            raise LLMError(f"Ollama HTTP error: {e}") from e
        except ValueError as e:  # JSON decode of envelope
            raise LLMError(f"Ollama returned non-JSON envelope: {e}") from e
```

```python
from __future__ import annotations

from call_monitor.db.models import Call, Profile

# Bump only when the prompt structure changes in a way that warrants
# re-ranking. Stored on every CallRanking row so historical scores stay
# attributable to the regime that produced them.
PROMPT_VERSION = "v1"

_SYSTEM_TEMPLATE = """\
You are a research-funding analyst for an academic hospital.
Score how relevant a funding call is to this profile.

PROFILE:
  Name: {name}
  Kind: {kind}
  Description: {description}
  Structured constraints:
    Areas: {areas}
    Budget range: {budget_range}
    TRL: {trl_range}
    Must include (force-include keywords): {must_include}
    Hard exclusions: {hard_exclusions}

Output a single JSON object with two keys:
  {{"score": <integer 0-100>, "rationale": "<2-3 sentences>"}}

Hard exclusions: if the call clearly hits one, score MUST be <= 10.
"""

_USER_TEMPLATE = """\
CALL:
  Title: {title}
  Programme: {programme}
  Deadline: {deadline}
  Budget: {budget}
  Summary: {summary}
"""


def _fmt_list(value) -> str:
    if not value:
        return "(none)"
    return ", ".join(str(v) for v in value)


def _budget_range(structured: dict) -> str:
    lo, hi = structured.get("budget_min"), structured.get("budget_max")
    if lo is None and hi is None:
        return "(unspecified)"
    return f"{lo or '—'}–{hi or '—'}"


def _trl_range(structured: dict) -> str:
    lo, hi = structured.get("trl_min"), structured.get("trl_max")
    if lo is None and hi is None:
        return "(unspecified)"
    return f"{lo or '—'}–{hi or '—'}"


def _summary_text(call: Call) -> str:
    """Prefer summary; fall back to first 2000 chars of full_text."""
    if call.summary:
        return call.summary
    if call.full_text:
        return call.full_text[:2000]
    return "(no summary)"


def build_messages(profile: Profile, call: Call) -> list[dict]:
    s = profile.structured or {}
    system = _SYSTEM_TEMPLATE.format(
        name=profile.name,
        kind=profile.kind.value if hasattr(profile.kind, "value") else profile.kind,
        description=profile.description or "(no description)",
        areas=_fmt_list(s.get("areas")),
        budget_range=_budget_range(s),
        trl_range=_trl_range(s),
        must_include=_fmt_list(s.get("must_include")),
        hard_exclusions=_fmt_list(s.get("hard_exclusions")),
    )

    deadline = (
        call.deadline.isoformat()
        if call.deadline is not None
        else "(no deadline)"
    )
    budget = (
        f"{call.budget_min or '—'}–{call.budget_max or '—'} {call.currency or ''}"
        if (call.budget_min or call.budget_max)
        else "(unspecified)"
    )
    user = _USER_TEMPLATE.format(
        title=call.title,
        programme=call.programme or "(unspecified)",
        deadline=deadline,
        budget=budget,
        summary=_summary_text(call),
    )

    return [
        {"role": "system", "content": system},
        {"role": "user", "content": user},
    ]
```

```python
from __future__ import annotations

import json
import logging
from dataclasses import dataclass

from call_monitor.db.models import Call, Profile
from call_monitor.ranking.ollama_client import LLMError, OllamaClient
from call_monitor.ranking.prompt import PROMPT_VERSION, build_messages

_logger = logging.getLogger(__name__)
_RATIONALE_MAX = 1000


@dataclass(slots=True)
class RankResult:
    score: int
    rationale: str
    model: str
    prompt_version: str
    raw_response: dict


def _extract_assistant_content(envelope: dict) -> str:
    try:
        return envelope["choices"][0]["message"]["content"]
    except (KeyError, IndexError, TypeError) as e:
        raise LLMError(f"Ollama envelope missing choices[0].message.content: {e}") from e


def _parse_score_rationale(content: str) -> tuple[int, str]:
    try:
        parsed = json.loads(content)
    except ValueError as e:
        raise LLMError(f"assistant content is invalid JSON: {e}") from e
    if not isinstance(parsed, dict):
        raise LLMError(f"assistant content is not a JSON object: {type(parsed).__name__}")
    if "score" not in parsed:
        raise LLMError("assistant content missing required key 'score'")
    score = parsed["score"]
    if not isinstance(score, int) or not 0 <= score <= 100:
        raise LLMError(f"score must be an integer in 0–100; got {score!r}")
    rationale = str(parsed.get("rationale") or "")
    if len(rationale) > _RATIONALE_MAX:
        rationale = rationale[:_RATIONALE_MAX]
    return score, rationale


def rank_pair(
    *, client: OllamaClient, profile: Profile, call: Call
) -> RankResult:
    """Score one (call, profile) pair via the LLM.

    Raises LLMError on any HTTP / parse / validation failure; the queue
    worker turns those into job retries (with backoff) and dead-letters
    after the attempt cap.
    """
    messages = build_messages(profile, call)
    envelope = client.chat(messages=messages)
    content = _extract_assistant_content(envelope)
    score, rationale = _parse_score_rationale(content)
    model = envelope.get("model") or client.model  # fallback for sparse responses
    return RankResult(
        score=score,
        rationale=rationale,
        model=model,
        prompt_version=PROMPT_VERSION,
        raw_response=envelope,
    )
```

```python
from __future__ import annotations

import logging
from datetime import UTC, datetime

from sqlalchemy import select, text, update
from sqlalchemy.dialects.postgresql import insert
from sqlalchemy.orm import Session

from call_monitor.db.models import RankingJob, RankingJobStatus

_logger = logging.getLogger(__name__)


def enqueue_for_pair(session: Session, *, call_id: int, profile_id: int) -> bool:
    """Insert a pending RankingJob for (call_id, profile_id).

    Idempotent: ON CONFLICT DO NOTHING against the partial unique index
    `uq_ranking_jobs_active_pair` makes the conflict path a silent no-op.
    Returns True on insert, False on skip.

    Why ON CONFLICT (not try/except IntegrityError + rollback): callers
    typically run inside a larger transaction (the orchestrator wraps
    upsert + enqueue in one). A `session.rollback()` here would discard
    the surrounding work too. ON CONFLICT keeps the failure scoped to
    this single statement and never fails the surrounding transaction.
    """
    stmt = (
        insert(RankingJob)
        .values(call_id=call_id, profile_id=profile_id, status=RankingJobStatus.PENDING)
        .on_conflict_do_nothing(
            index_elements=["call_id", "profile_id"],
            index_where=text("status IN ('pending', 'running')"),
        )
    )
    result = session.execute(stmt)
    # rowcount is 1 on insert, 0 on conflict-skip. Postgres returns this
    # reliably for ON CONFLICT.
    return result.rowcount > 0


def claim_pending_jobs(session: Session, *, limit: int) -> list[RankingJob]:
    """Atomically transition up to `limit` pending jobs into 'running' and
    return them. Uses SELECT FOR UPDATE SKIP LOCKED so concurrent workers
    (none today, but the queue model is forward-compatible) don't fight
    over the same rows.
    """
    stmt = (
        select(RankingJob)
        .where(RankingJob.status == RankingJobStatus.PENDING)
        .order_by(RankingJob.updated_at.asc())
        .limit(limit)
        .with_for_update(skip_locked=True)
    )
    rows = session.execute(stmt).scalars().all()
    if not rows:
        return []
    now = datetime.now(UTC)
    for row in rows:
        row.status = RankingJobStatus.RUNNING
        row.updated_at = now
    session.flush()
    return rows


def mark_done(session: Session, job_id: int) -> None:
    session.execute(
        update(RankingJob)
        .where(RankingJob.id == job_id)
        .values(status=RankingJobStatus.DONE, updated_at=datetime.now(UTC))
    )


def mark_failed(
    session: Session, job_id: int, *, error: str, terminal: bool
) -> None:
    """Mark a job failed. If terminal=True, transition to the FAILED
    terminal status (dead-letter). Otherwise revert to PENDING with
    incremented attempts so the queue retries it on the next tick."""
    new_status = RankingJobStatus.FAILED if terminal else RankingJobStatus.PENDING
    session.execute(
        update(RankingJob)
        .where(RankingJob.id == job_id)
        .values(
            status=new_status,
            attempts=RankingJob.attempts + 1,
            last_error=error,
            updated_at=datetime.now(UTC),
        )
    )
```

```python
from __future__ import annotations

import logging
from collections.abc import Callable

from sqlalchemy import select
from sqlalchemy.orm import sessionmaker

from call_monitor.db.models import Call, CallRanking, Profile, RankingJob
from call_monitor.ranking.ollama_client import LLMError, OllamaClient
from call_monitor.ranking.queue import claim_pending_jobs, mark_done, mark_failed
from call_monitor.ranking.rank_pair import RankResult, rank_pair as default_rank_pair

_logger = logging.getLogger(__name__)

# Per spec: 5 attempts before dead-letter.
#
# Spec deviation (intentional): the spec describes exponential backoff
# (30s, 2m, 10m). v1 ships with FLAT retries — a failed job goes back to
# PENDING and the next scheduler tick (60s default) retries it. So the
# effective wait is one tick per attempt, regardless of attempt index.
# Acceptable at v1 because failure rates are low (Ollama is local; the
# JSON-mode contract eliminates parse churn). True exponential backoff
# would require a `next_attempt_at` column on ranking_jobs and a WHERE
# clause on `claim_pending_jobs` — straightforward to add when needed.
MAX_ATTEMPTS = 5

# Type for an injectable rank function (real one is rank_pair from
# rank_pair.py; tests inject a static double).
Ranker = Callable[..., RankResult]


def drain_queue(
    *,
    session_factory: sessionmaker,
    client: OllamaClient | None,
    ranker: Ranker = default_rank_pair,
    batch_size: int = 10,
) -> int:
    """Claim up to `batch_size` pending jobs, rank each, persist, mark
    done-or-failed. Returns the number of jobs processed (whether or not
    they succeeded).

    `client` is the OllamaClient. It's optional in tests where `ranker`
    is a static double that doesn't call the LLM.
    """
    with session_factory() as session:
        jobs = claim_pending_jobs(session, limit=batch_size)
        if not jobs:
            return 0
        # Capture identifying fields before commit + close — we'll re-open
        # per-job so a single bad job doesn't poison the whole batch.
        claimed = [
            (job.id, job.call_id, job.profile_id, job.attempts) for job in jobs
        ]
        session.commit()

    processed = 0
    for job_id, call_id, profile_id, attempts in claimed:
        processed += 1
        try:
            with session_factory() as session:
                call = session.get(Call, call_id)
                profile = session.get(Profile, profile_id)
                if call is None or profile is None:
                    # Either deleted between enqueue and run — dead-letter.
                    mark_failed(
                        session,
                        job_id,
                        error="call or profile vanished",
                        terminal=True,
                    )
                    session.commit()
                    continue
                result = ranker(client=client, profile=profile, call=call)
                session.merge(
                    CallRanking(
                        call_id=call_id,
                        profile_id=profile_id,
                        score=result.score,
                        rationale=result.rationale,
                        model=result.model,
                        prompt_version=result.prompt_version,
                        raw_response=result.raw_response,
                    )
                )
                mark_done(session, job_id)
                session.commit()
        except LLMError as e:
            terminal = (attempts + 1) >= MAX_ATTEMPTS
            with session_factory() as session:
                mark_failed(session, job_id, error=str(e), terminal=terminal)
                session.commit()
            _logger.warning(
                "ranking.job_failed",
                extra={
                    "event": "ranking_job_failed",
                    "job_id": job_id,
                    "attempts": attempts + 1,
                    "terminal": terminal,
                    "error": str(e),
                },
            )
        except Exception as e:  # noqa: BLE001 — worker must not crash the scheduler
            terminal = (attempts + 1) >= MAX_ATTEMPTS
            with session_factory() as session:
                mark_failed(
                    session, job_id, error=f"{type(e).__name__}: {e}", terminal=terminal
                )
                session.commit()
            _logger.exception(
                "ranking.job_unexpected",
                extra={
                    "event": "ranking_job_unexpected",
                    "job_id": job_id,
                    "attempts": attempts + 1,
                    "terminal": terminal,
                },
            )
    return processed
```

```python
from __future__ import annotations

import logging
from datetime import UTC, datetime, timedelta

from sqlalchemy import select
from sqlalchemy.orm import sessionmaker

from call_monitor.db.models import Call, Profile
from call_monitor.ranking.queue import enqueue_for_pair

_logger = logging.getLogger(__name__)
BACKFILL_DAYS = 90


def backfill_profile(*, session_factory: sessionmaker, profile_id: int) -> int:
    """Enqueue ranking jobs for every Call from the last BACKFILL_DAYS × this
    profile. Returns the number of jobs newly enqueued (existing pending/
    running jobs are skipped via the partial unique index).
    """
    with session_factory() as session:
        profile = session.get(Profile, profile_id)
        if profile is None or not profile.is_active:
            return 0
        cutoff = datetime.now(UTC) - timedelta(days=BACKFILL_DAYS)
        call_ids = session.execute(
            select(Call.id).where(Call.published_at >= cutoff)
        ).scalars().all()
        enqueued = 0
        for cid in call_ids:
            if enqueue_for_pair(session, call_id=cid, profile_id=profile_id):
                enqueued += 1
        session.commit()
    _logger.info(
        "ranking.backfill_done",
        extra={
            "event": "ranking_backfill_done",
            "profile_id": profile_id,
            "candidate_calls": len(call_ids),
            "enqueued": enqueued,
        },
    )
    return enqueued
```

```python
from sqlalchemy import select
from sqlalchemy.orm import joinedload

from call_monitor.db.models import Call, CallRanking, Profile, Source, User


@router.get("/", response_class=HTMLResponse)
def calls_index(
    request: Request,
    source: str | None = Query(default=None),
    sort: _Sort = Query(default="newest"),  # noqa: B008
    profile: int | None = Query(default=None),
    limit: int = Query(default=50, ge=1, le=200),
    user: User = Depends(get_current_user),  # noqa: B008
):
    session_factory = request.app.state.session_factory
    with session_factory() as session:
        # When a profile is selected, LEFT JOIN call_rankings for THAT
        # profile and sort by score desc (nulls last). Without a profile,
        # fall back to the unranked sort (newest/deadline).
        if profile is not None:
            stmt = (
                select(Call, CallRanking)
                .options(joinedload(Call.source))
                .outerjoin(
                    CallRanking,
                    (CallRanking.call_id == Call.id)
                    & (CallRanking.profile_id == profile),
                )
                .order_by(
                    CallRanking.score.desc().nullslast(),
                    Call.published_at.desc().nullslast(),
                    Call.id.desc(),
                )
                .limit(limit)
            )
        else:
            stmt = (
                select(Call, None)
                .options(joinedload(Call.source))
                .order_by(*_order_for(sort))
                .limit(limit)
            )
        if source:
            stmt = stmt.where(
                Call.source_id == select(Source.id).where(Source.slug == source).scalar_subquery()
            )
        rows = session.execute(stmt).unique().all()
        # Normalize to (Call, CallRanking|None) shape regardless of branch
        calls_with_rankings: list[tuple[Call, CallRanking | None]] = [
            (row[0], row[1] if profile is not None else None) for row in rows
        ]
        sources = session.execute(select(Source).order_by(Source.name)).scalars().all()
        profiles = session.execute(
            select(Profile).where(Profile.is_active.is_(True)).order_by(Profile.name)
        ).scalars().all()
        session.expunge_all()

    return request.app.state.templates.TemplateResponse(
        request=request,
        name="calls_index.html",
        context={
            "calls_with_rankings": calls_with_rankings,
            "sources": sources,
            "profiles": profiles,
            "selected_source": source,
            "selected_sort": sort,
            "selected_profile_id": profile,
            "user": user,
        },
    )
```

```python
@router.get("/admin", response_class=HTMLResponse)
def admin_index(
    request: Request,
    user: User = Depends(require_admin),  # noqa: B008
):
    session_factory = request.app.state.session_factory
    with session_factory() as session:
        rows = session.execute(
            select(Source, func.count(Call.id).label("call_count"))
            .outerjoin(Call, Call.source_id == Source.id)
            .group_by(Source.id)
            .order_by(Source.name)
        ).all()
        sources = [
            {
                "name": src.name,
                "slug": src.slug,
                "last_fetched_at": src.last_fetched_at,
                "call_count": cc,
            }
            for src, cc in rows
        ]
        # Ranking-queue counts by status
        queue_rows = session.execute(
            select(RankingJob.status, func.count(RankingJob.id))
            .group_by(RankingJob.status)
        ).all()
        queue_counts = {status: count for status, count in queue_rows}

    return request.app.state.templates.TemplateResponse(
        request=request,
        name="admin.html",
        context={
            "sources": sources,
            "queue_counts": queue_counts,
            "queue_status_pending": RankingJobStatus.PENDING.value,
            "queue_status_running": RankingJobStatus.RUNNING.value,
            "queue_status_failed": RankingJobStatus.FAILED.value,
            "queue_status_done": RankingJobStatus.DONE.value,
            "user": user,
        },
    )

def admin_rerank(
    request: Request,
    user: User = Depends(require_admin),  # noqa: B008
```
